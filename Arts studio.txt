    <!-- Firebase SDKs -->
    <script type="module">
        // Import the functions you need from the SDKs you need
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getAuth, signInAnonymously, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { getFirestore, doc, getDoc, setDoc, collection, addDoc, serverTimestamp, runTransaction, onSnapshot, setLogLevel, enableIndexedDbPersistence } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        const firebaseConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : {};
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';

        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);
        setLogLevel('debug');

        // --- GEMINI API ---
        const apiKey = ""; // Provided by the environment
        const model = 'gemini-2.5-flash-preview-05-20';
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`;

        async function callGemini(prompt, retries = 3, delay = 1000) {
            try {
                const payload = { contents: [{ parts: [{ text: prompt }] }] };
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                if (!response.ok) {
                     if (response.status === 429 && retries > 0) {
                        await new Promise(res => setTimeout(res, delay));
                        return callGemini(prompt, retries - 1, delay * 2); // Exponential backoff
                    }
                    throw new Error(`API call failed with status: ${response.status}`);
                }
                const result = await response.json();
                return result.candidates?.[0]?.content?.parts?.[0]?.text || "No response from the wizard.";
            } catch (error) {
                console.error("Gemini API call failed:", error);
                if (retries > 0) {
                    await new Promise(res => setTimeout(res, delay));
                    return callGemini(prompt, retries - 1, delay * 2);
                }
                return "The wizard is currently resting. Please try again later.";
            }
        }
        
        // --- OFFLINE PERSISTENCE ---
        try {
            await enableIndexedDbPersistence(db);
            console.log("Offline persistence enabled.");
            document.getElementById('offlineStatus').textContent = 'Offline Mode Ready';
        } catch (err) {
            console.warn("Firestore persistence failed:", err.message);
        }

        let currentUserId = null;
        let userDocRef = null;

        // --- AUTHENTICATION & DATA LISTENERS ---
        onAuthStateChanged(auth, async (user) => {
            if (user) {
                currentUserId = user.uid;
                document.getElementById('userIdDisplay').textContent = `User ID: ${currentUserId}`;
                
                const profileCollectionPath = `artifacts/${appId}/users/${currentUserId}/profileData`;
                userDocRef = doc(db, profileCollectionPath, "main");

                const userDoc = await getDoc(userDocRef);
                if (!userDoc.exists()) {
                    await setDoc(userDocRef, { gwaBalance: 1250, createdAt: serverTimestamp() });
                }
                
                onSnapshot(userDocRef, (doc) => {
                    document.getElementById('gwa-balance').textContent = doc.exists() ? (doc.data().gwaBalance || 0) : 0;
                });

                const submissionsCollectionPath = `artifacts/${appId}/users/${currentUserId}/submissions`;
                const submissionsQuery = collection(db, submissionsCollectionPath);
                
                onSnapshot(submissionsQuery, (snapshot) => {
                    const logContainer = document.getElementById('submissionLog');
                    logContainer.innerHTML = ''; 

                    if (snapshot.empty) {
                        logContainer.innerHTML = '<p class="text-slate-500">No submissions yet...</p>';
                        return;
                    }
                    
                    const submissions = snapshot.docs.map(doc => doc.data());
                    submissions.sort((a, b) => (b.createdAt?.seconds || 0) - (a.createdAt?.seconds || 0));

                    submissions.forEach(submission => {
                        const statusColor = submission.status === 'pending_review' ? 'text-yellow-400' : 'text-green-400';
                        const title = submission.type === 'commercial' ? submission.title : submission.task;

                        const logEntry = document.createElement('div');
                        logEntry.className = 'bg-slate-800/50 p-3 rounded-lg';
                        logEntry.innerHTML = `
                            <p class="font-semibold text-sm truncate">${title}</p>
                            <div class="flex justify-between items-center mt-1">
                                <p class="text-xs text-slate-400">${submission.artType || submission.project}</p>
                                <p class="text-xs font-medium ${statusColor}">${submission.status.replace('_', ' ')}</p>
                            </div>
                        `;
                        logContainer.appendChild(logEntry);
                    });
                });

                fetchDailyDirective(); // Fetch initial directive on login
            } else {
                signInAnonymously(auth).catch((error) => console.error("Anonymous sign-in failed:", error));
            }
        });
        
        // --- GEMINI-POWERED FEATURES ---
        window.fetchDailyDirective = async function() {
            const contentDiv = document.getElementById('directivesContent');
            const button = document.getElementById('newDirectiveBtn');
            button.disabled = true;
            contentDiv.innerHTML = '<p class="text-slate-400 text-sm">The wizard is conjuring a new directive...</p>';

            const prompt = "Generate a single, short, inspiring creative directive (2-3 sentences max) for a community of digital artists and creators. Be mystical and encouraging, like a wise wizard guiding their apprentices. Frame it as a 'Directive'.";
            const directiveText = await callGemini(prompt);

            contentDiv.innerHTML = `
                <div class="bg-slate-800/50 p-3 rounded-lg">
                    <p class="font-semibold text-sm">Today's Directive <span class="text-xs text-slate-500 float-right">Just Now</span></p>
                    <p class="text-xs text-slate-300 mt-1">${directiveText}</p>
                </div>`;
            button.disabled = false;
        }

        document.getElementById('ideaGeneratorForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const artType = document.getElementById('ideaArtType').value;
            const theme = document.getElementById('ideaTheme').value;
            const resultsDiv = document.getElementById('ideaResults');
            const spinner = document.getElementById('ideaGenSpinner');
            const btnText = document.getElementById('ideaGenBtnText');
            
            spinner.classList.remove('hidden');
            btnText.textContent = "Generating...";
            resultsDiv.textContent = 'Thinking...';

            let themeClause = theme ? ` based on the theme: "${theme}"` : "";
            const prompt = `You are a creative wizard. Generate 5 distinct and actionable project ideas for a digital artist creating a "${artType}"${themeClause}. Format them as a numbered list.`;
            
            const ideas = await callGemini(prompt);
            resultsDiv.textContent = ideas;
            spinner.classList.add('hidden');
            btnText.textContent = "Generate Ideas";
        });


        // --- GLOBAL UI FUNCTIONS ---
        window.showModal = function(modalId) {
            document.getElementById(modalId).classList.add('flex');
            document.getElementById(modalId).classList.remove('hidden');
        }

        window.closeModal = function(modalId) {
            document.getElementById(modalId).classList.remove('flex');
            document.getElementById(modalId).classList.add('hidden');
        }

        function showSuccessMessage(message) {
             document.getElementById('successMessage').textContent = message;
             showModal('successModal');
        }

        // --- FIRESTORE ACTIONS ---
        document.getElementById('commercialArtForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            if (!currentUserId) return;
            const title = document.getElementById('artTitle').value;
            const type = document.getElementById('artType').value;

            try {
                const submissionsCollectionPath = `artifacts/${appId}/users/${currentUserId}/submissions`;
                await addDoc(collection(db, submissionsCollectionPath), {
                    userId: currentUserId, type: 'commercial', title, artType: type, status: 'pending_review', createdAt: serverTimestamp()
                });
                closeModal('commercialArtModal');
                showSuccessMessage('Your Art has been submitted for review.');
                e.target.reset();
            } catch (error) {
                console.error("Error adding document: ", error);
            }
        });

        document.getElementById('communityContributionForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            if (!currentUserId) return;
            const task = document.getElementById('contributionTask').value;
            const project = document.getElementById('contributionProject').value;
            try {
                const submissionsCollectionPath = `artifacts/${appId}/users/${currentUserId}/submissions`;
                await addDoc(collection(db, submissionsCollectionPath), {
                    userId: currentUserId, type: 'community', task, project, status: 'pending_review', createdAt: serverTimestamp()
                });
                closeModal('communityContributionModal');
                showSuccessMessage('Your contribution has been logged.');
                e.target.reset();
            } catch (error) {
                console.error("Error adding document: ", error);
            }
        });

        window.redeemGWA = async function() {
            const amountInput = document.getElementById('redeemAmount');
            const amount = parseInt(amountInput.value, 10);

            if (!currentUserId || isNaN(amount) || amount <= 0) return;

            try {
                await runTransaction(db, async (transaction) => {
                    const userDoc = await transaction.get(userDocRef);
                    if (!userDoc.exists()) throw "User document does not exist!";
                    const currentBalance = userDoc.data().gwaBalance || 0;
                    if (amount > currentBalance) throw "Insufficient GWA balance.";
                    
                    transaction.update(userDocRef, { gwaBalance: currentBalance - amount });
                });
                amountInput.value = '';
                showSuccessMessage(`${amount} GWA successfully redeemed!`);
            } catch (e) {
                console.error("Redemption failed: ", e);
                showSuccessMessage(typeof e === 'string' ? e : 'Redemption failed.');
            }
        }
    </script>
